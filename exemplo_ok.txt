int soma(int a, int b) {
    return a + b;
}

int principal(int limite) {
    int contador = 0;
    int total = 0;

    while (contador < limite) {
        total = total + soma(contador, 1);
        if (total >= 10) {
            break;
        } elif (contador == 3) {
            int extra = 2;
            total = total + extra;
        } else {
            int contador = 99;
            total = total + 1;
        }
        contador = contador + 1;
    }

    return total;
}
