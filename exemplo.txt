num principal(num limite) {
    num contador = 0;
    num total = 0;

    while (contador < limite) {
        total = total + soma(contador, 1);

        if (total >= 10) {
            break;
        } elif (contador == 3) {
            total = total + 2;
        } else {
            total = total + 1;
        }

        contador = contador + 1;
    }

    return total;
}
